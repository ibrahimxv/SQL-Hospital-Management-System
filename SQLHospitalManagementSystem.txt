CREATE DATABASE HOSPITALMANAGEMENT

USE HOSPITALMANAGEMENT

CREATE TABLE Patients(
Id INT PRIMARY KEY IDENTITY,
FirstName NVARCHAR(20) NOT NULL,
LastName NVARCHAR (25) NOT NULL,
BirthDate DATE NOT NULL CHECK (BirthDate <= CAST(GETDATE() AS DATE)),
Email VARCHAR NOT NULL UNIQUE,
)

ALTER TABLE Patients ALTER COLUMN Email VARCHAR(100) NOT NULL

CREATE TABLE Departments(
Id INT PRIMARY KEY IDENTITY,
Name VARCHAR (40) NOT NULL UNIQUE,
)

CREATE TABLE Doctors(
Id INT PRIMARY KEY IDENTITY,
FirstName NVARCHAR(20) NOT NULL,
LastName NVARCHAR (25) NOT NULL,
Experience INT NOT NULL CHECK (Experience >= 0),
)

CREATE TABLE DoctorDepartments(
DoctorId INT,
DepartmentId INT,
PRIMARY KEY (DoctorID, DepartmentId),
FOREIGN KEY (DoctorID) REFERENCES Doctors(Id),
FOREIGN KEY (DepartmentId) REFERENCES Departments(Id)
)

CREATE TABLE DoctorPatients(
DoctorId INT,
PatientId INT,
PRIMARY KEY (DoctorId, PatientId),
FOREIGN KEY (DoctorId) REFERENCES Doctors(Id),
FOREIGN KEY (PatientId) REFERENCES Patients(Id)
)

SELECT Doctors.FirstName, Doctors.LastName, Departments.Name AS Department
FROM Doctors
JOIN DoctorDepartments
ON Doctors.Id = DoctorDepartments.DoctorId
JOIN Departments
ON Departments.Id = DoctorDepartments.DepartmentId

SELECT 
Patients.FirstName AS PatientFirstName,
Patients.LastName AS PatientLastName,
Doctors.FirstName AS DoctorFirstName,
Doctors.LastName AS DoctorLastName
FROM Patients
JOIN DoctorPatients
ON Patients.Id = DoctorPatients.PatientId
JOIN Doctors
ON DoctorPatients.DoctorId = Doctors.Id

CREATE TABLE Appointments(
Id INT PRIMARY KEY IDENTITY,
PatientId INT NOT NULL,
DoctorId INT NOT NULL,
AppointmenDate DATETIME NOT NULL,
Result VARCHAR (200),

FOREIGN KEY (DoctorId) REFERENCES Doctors(Id),
FOREIGN KEY (PatientId) REFERENCES Patients(Id)
)

CREATE TABLE AppointmentMedicines(
	AppointmentId INT,
	MedicineId INT,

	PRIMARY KEY (AppointmentId, MedicineId),

	FOREIGN KEY (AppointmentId) REFERENCES Appointments(Id),
	FOREIGN KEY (MedicineId) REFERENCES Medicine(Id)
)

SELECT
Patients.FirstName AS PatientFirstName,
Patients.LastName AS PatientLastName,
Medicine.Name AS MedicineName
FROM Patients
JOIN Appointments 
ON Patients.Id = Appointments.PatientId
JOIN AppointmentMedicines
ON Appointments.Id = AppointmentMedicines.AppointmentId
JOIN Medicine
ON AppointmentMedicines.MedicineId = Medicine.Id
